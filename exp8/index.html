// Task 3: Book catalogue API with filtering, validation and consistent errors
require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const Book = require("./models/Book");

const app = express();
app.use(express.json());

// Consistent error format: { "error": { "status": 400, "message": "...", "details": [...] } }
const sendError = (res, status, message, details) =>
  res.status(status).json({ error: { status, message, ...(details && { details }) } });

// Escape user text before using it in a regular expression
const escapeRegex = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

// Validate incoming book data
function validateBook(body) {
  const errors = [];
  const { title, author, publicationYear, availableCopies } = body;
  if (typeof title !== "string" || title.trim() === "") errors.push("title must not be empty");
  if (typeof author !== "string" || author.trim() === "") errors.push("author must not be empty");
  const year = Number(publicationYear);
  if (publicationYear === undefined || !Number.isInteger(year) || year < 1000 || year > new Date().getFullYear())
    errors.push(`publicationYear must be a valid year (1000-${new Date().getFullYear()})`);
  const copies = Number(availableCopies);
  if (availableCopies === undefined || !Number.isFinite(copies) || copies < 0)
    errors.push("availableCopies must be a non-negative number");
  return errors;
}

// GET /books?category=Science&author=Rao  (both filters optional, case-insensitive)
app.get("/books", async (req, res, next) => {
  try {
    const filter = {};
    if (req.query.category) filter.category = new RegExp(`^${escapeRegex(req.query.category)}$`, "i");
    if (req.query.author)   filter.author   = new RegExp(escapeRegex(req.query.author), "i");
    res.status(200).json(await Book.find(filter));
  } catch (err) { next(err); }
});

// GET /books/:id
app.get("/books/:id", async (req, res, next) => {
  if (!mongoose.Types.ObjectId.isValid(req.params.id)) return sendError(res, 400, "Invalid book id");
  try {
    const book = await Book.findById(req.params.id);
    if (!book) return sendError(res, 404, "Book not found");
    res.status(200).json(book);
  } catch (err) { next(err); }
});

// POST /books
app.post("/books", async (req, res, next) => {
  const errors = validateBook(req.body);
  if (errors.length) return sendError(res, 400, "Validation failed", errors);
  try {
    const { title, author, category, publicationYear, availableCopies } = req.body;
    const book = await Book.create({ title, author, category, publicationYear, availableCopies });
    res.status(201).json(book);
  } catch (err) { next(err); }
});

// Unknown route -> 404 in the same format
app.use((req, res) => sendError(res, 404, `Route ${req.method} ${req.originalUrl} not found`));

// Invalid JSON body and unexpected errors
app.use((err, req, res, next) => {
  if (err.type === "entity.parse.failed") return sendError(res, 400, "Invalid JSON in request body");
  console.error(err);                                    // log details server-side only
  sendError(res, 500, "Internal server error");
});

mongoose.connect(process.env.MONGO_URI)
  .then(() => {
    console.log("MongoDB connected");
    app.listen(process.env.PORT || 3001, () => console.log("book-api running"));
  })
  .catch((err) => { console.error("MongoDB connection failed:", err.message); process.exit(1); });

// Task 2: CRUD operations with MongoDB (Mongoose)
require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const Course = require("./models/Course");

const app = express();
app.use(express.json());

const isValidId = (id) => mongoose.Types.ObjectId.isValid(id);

// Convert Mongoose errors to a status code + message
function handleError(res, err) {
  if (err.name === "ValidationError") return res.status(400).json({ error: Object.values(err.errors).map((e) => e.message).join(", ") });
  if (err.code === 11000) return res.status(400).json({ error: "courseCode already exists" });
  console.error(err);                                    // details stay in the server log only
  return res.status(500).json({ error: "Internal server error" });
}

// CREATE
app.post("/courses", async (req, res) => {
  try {
    const course = await Course.create(req.body);
    res.status(201).json(course);
  } catch (err) { handleError(res, err); }
});

// READ all
app.get("/courses", async (req, res) => {
  try {
    res.status(200).json(await Course.find());
  } catch (err) { handleError(res, err); }
});

// READ one
app.get("/courses/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid course id" });
  try {
    const course = await Course.findById(req.params.id);
    if (!course) return res.status(404).json({ error: "Course not found" });
    res.status(200).json(course);
  } catch (err) { handleError(res, err); }
});

// UPDATE
app.put("/courses/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid course id" });
  try {
    const course = await Course.findByIdAndUpdate(req.params.id, req.body, { new: true, runValidators: true });
    if (!course) return res.status(404).json({ error: "Course not found" });
    res.status(200).json(course);
  } catch (err) { handleError(res, err); }
});

// DELETE
app.delete("/courses/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid course id" });
  try {
    const course = await Course.findByIdAndDelete(req.params.id);
    if (!course) return res.status(404).json({ error: "Course not found" });
    res.status(200).json({ message: "Course deleted", id: course._id });
  } catch (err) { handleError(res, err); }
});

mongoose.connect(process.env.MONGO_URI)
  .then(() => {
    console.log("MongoDB connected");
    app.listen(process.env.PORT || 3000, () => console.log("course-api running"));
  })
  .catch((err) => { console.error("MongoDB connection failed:", err.message); process.exit(1); });

// Task 1: Basic REST API for student records (in-memory array)
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());   // parse JSON request bodies into req.body

let students = [
  { id: 1, name: "Aarav Shah",  department: "EXTC", semester: 5 },
  { id: 2, name: "Riya Mehta",  department: "COMP", semester: 5 },
  { id: 3, name: "Kabir Nair",  department: "EXTC", semester: 3 },
  { id: 4, name: "Sneha Patil", department: "IT",   semester: 7 },
  { id: 5, name: "Dev Joshi",   department: "COMP", semester: 1 },
];
let nextId = 6;

// GET /students -> all students (200)
app.get("/students", (req, res) => {
  res.status(200).json(students);
});

// GET /students/:id -> one student (200) or 404
app.get("/students/:id", (req, res) => {
  const student = students.find((s) => s.id === Number(req.params.id));
  if (!student) return res.status(404).json({ error: "Student not found" });
  res.status(200).json(student);
});

// POST /students -> create (201) or 400 when fields are missing
app.post("/students", (req, res) => {
  const { name, department, semester } = req.body;
  if (!name || !department || semester === undefined) {
    return res.status(400).json({ error: "name, department and semester are required" });
  }
  const student = { id: nextId++, name, department, semester: Number(semester) };
  students.push(student);
  res.status(201).json(student);
});

app.listen(PORT, () => console.log(`student-api running on http://localhost:${PORT}`));
