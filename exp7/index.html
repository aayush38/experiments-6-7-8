models/Application.js

const mongoose = require("mongoose");

// Student application (admission) form
const applicationSchema = new mongoose.Schema(
  {
    fullName: { type: String, required: [true, "Full name is required"], trim: true },
    email:    { type: String, required: [true, "Email is required"], trim: true, lowercase: true,
                match: [/^\S+@\S+\.\S+$/, "Enter a valid email"] },
    phone:    { type: String, required: true, match: [/^[0-9]{10}$/, "Phone must be 10 digits"] },
    dob:      { type: Date, required: [true, "Date of birth is required"] },
    gender:   { type: String, enum: ["Male", "Female", "Other"], required: true },
    course:   { type: String, required: true, enum: ["Computer Engineering", "Electronics and Computer Engineering", "Information Technology"] },
    percentage: { type: Number, required: true, min: 0, max: 100 },
    address:  { type: String, trim: true },
  },
  { timestamps: true }
);

module.exports = mongoose.model("Application", applicationSchema);

models/Student.js

const mongoose = require("mongoose");

// Schema describes the structure and validation rules of a Student document
const studentSchema = new mongoose.Schema(
  {
    rollNo:   { type: Number, required: [true, "Roll number is required"], unique: true },
    name:     { type: String, required: [true, "Name is required"], trim: true },
    dept:     { type: String, required: true, enum: ["COMP", "EXTC", "IT", "MECH"] },
    semester: { type: Number, required: true, min: 1, max: 8 },
    cgpa:     { type: Number, min: 0, max: 10, default: 0 },
  },
  { timestamps: true }
);

module.exports = mongoose.model("Student", studentSchema);

db.students.drop();

// ---- 1. Insert documents
db.students.insertMany([
  { rollNo: 1, name: "Aarav Shah",  dept: "EXTC", semester: 5, cgpa: 8.9, city: "Mumbai" },
  { rollNo: 2, name: "Riya Mehta",  dept: "COMP", semester: 5, cgpa: 9.2, city: "Pune"   },
  { rollNo: 3, name: "Kabir Nair",  dept: "EXTC", semester: 5, cgpa: 7.4, city: "Mumbai" },
  { rollNo: 4, name: "Sneha Patil", dept: "COMP", semester: 3, cgpa: 8.1, city: "Nashik" },
  { rollNo: 5, name: "Dev Joshi",   dept: "IT",   semester: 5, cgpa: 6.8, city: "Thane"  },
  { rollNo: 6, name: "Meera Iyer",  dept: "EXTC", semester: 3, cgpa: 9.5, city: "Mumbai" }
]);

// ---- 2. Find (read)
db.students.find();                                   // all documents
db.students.find({ dept: "EXTC" });                   // equality
db.students.findOne({ rollNo: 4 });                   // single document

// ---- 3. Comparison operators
db.students.find({ cgpa: { $gt: 8 } });               // greater than
db.students.find({ cgpa: { $gte: 8.9 } });            // greater than or equal
db.students.find({ cgpa: { $lt: 7.5 } });             // less than
db.students.find({ dept: { $ne: "EXTC" } });          // not equal
db.students.find({ city: { $in: ["Pune", "Nashik"] } });   // in list
db.students.find({ semester: { $nin: [3] } });        // not in list

// ---- 4. Logical operators
db.students.find({ $and: [ { dept: "EXTC" }, { cgpa: { $gt: 9 } } ] });
db.students.find({ $or:  [ { dept: "IT" }, { cgpa: { $gt: 9 } } ] });
db.students.find({ cgpa: { $not: { $gt: 8 } } });
db.students.find({ $nor: [ { dept: "COMP" }, { city: "Mumbai" } ] });

// ---- 5. Projection (1 = include, 0 = exclude)
db.students.find({}, { name: 1, cgpa: 1, _id: 0 });
db.students.find({ dept: "EXTC" }, { _id: 0, city: 0 });

// ---- 6. Sort, limit, count
db.students.find({}, { name: 1, cgpa: 1, _id: 0 }).sort({ cgpa: -1 }).limit(3);
db.students.countDocuments({ dept: "EXTC" });

// ---- 7. Update
db.students.updateOne({ rollNo: 5 }, { $set: { cgpa: 7.2, city: "Mumbai" } });
db.students.updateMany({ semester: 3 }, { $inc: { semester: 1 } });   // semester 3 -> 4
db.students.updateOne({ rollNo: 7 }, { $set: { name: "Isha Rao", dept: "IT", semester: 5, cgpa: 8.0, city: "Pune" } }, { upsert: true });

// ---- 8. Delete
db.students.deleteOne({ rollNo: 7 });
db.students.deleteMany({ cgpa: { $lt: 7.5 } });
db.students.find({}, { _id: 0, rollNo: 1, name: 1, cgpa: 1 });

routes/students.js

const express = require("express");
const mongoose = require("mongoose");
const Student = require("../models/Student");
const router = express.Router();

const isValidId = (id) => mongoose.Types.ObjectId.isValid(id);

// CREATE
router.post("/", async (req, res) => {
  try {
    const student = await Student.create(req.body);
    res.status(201).json(student);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// READ all
router.get("/", async (req, res) => {
  try {
    res.json(await Student.find().sort({ rollNo: 1 }));
  } catch (err) {
    res.status(500).json({ error: "Server error" });
  }
});

// READ one
router.get("/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid id" });
  try {
    const student = await Student.findById(req.params.id);
    if (!student) return res.status(404).json({ error: "Student not found" });
    res.json(student);
  } catch (err) {
    res.status(500).json({ error: "Server error" });
  }
});

// UPDATE
router.put("/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid id" });
  try {
    const student = await Student.findByIdAndUpdate(req.params.id, req.body, {
      new: true, runValidators: true,
    });
    if (!student) return res.status(404).json({ error: "Student not found" });
    res.json(student);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// DELETE
router.delete("/:id", async (req, res) => {
  if (!isValidId(req.params.id)) return res.status(400).json({ error: "Invalid id" });
  try {
    const student = await Student.findByIdAndDelete(req.params.id);
    if (!student) return res.status(404).json({ error: "Student not found" });
    res.json({ message: "Student deleted", id: student._id });
  } catch (err) {
    res.status(500).json({ error: "Server error" });
  }
});

module.exports = router;

server.js

require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const path = require("path");
const Application = require("./models/Application");
const studentRoutes = require("./routes/students");

const app = express();
const PORT = process.env.PORT || 3000;

// Middleware
app.use(express.json());                          // JSON bodies (REST API)
app.use(express.urlencoded({ extended: true }));  // HTML form bodies
app.use(express.static(path.join(__dirname, "public")));

// CRUD API for students
app.use("/api/students", studentRoutes);

const escapeHtml = (s) =>
  String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));

// Student application form: save to MongoDB
app.post("/apply", async (req, res) => {
  try {
    const saved = await Application.create(req.body);
    res.status(201).send(`
      <h2>Application submitted successfully</h2>
      <p>Application ID: <b>${saved._id}</b></p>
      <p>Thank you, ${escapeHtml(saved.fullName)}.</p>
      <a href="/applications">View all applications</a> | <a href="/apply.html">Submit another</a>`);
  } catch (err) {
    if (err.name === "ValidationError") {
      const msgs = Object.values(err.errors).map((e) => `<li>${escapeHtml(e.message)}</li>`).join("");
      return res.status(400).send(`<h2>Please fix the following</h2><ul>${msgs}</ul><a href="/apply.html">Go back</a>`);
    }
    res.status(500).send("Server error");
  }
});

// List all applications
app.get("/applications", async (req, res) => {
  try {
    const apps = await Application.find().sort({ createdAt: -1 });
    const rows = apps.map((a) => `<tr><td>${escapeHtml(a.fullName)}</td><td>${escapeHtml(a.email)}</td>
      <td>${escapeHtml(a.phone)}</td><td>${escapeHtml(a.course)}</td><td>${a.percentage}</td></tr>`).join("");
    res.send(`<h2>Student Applications (${apps.length})</h2>
      <table border="1" cellpadding="6"><tr><th>Name</th><th>Email</th><th>Phone</th><th>Course</th><th>%</th></tr>${rows}</table>
      <p><a href="/apply.html">New application</a></p>`);
  } catch (err) {
    res.status(500).send("Server error");
  }
});

// Connect to MongoDB, then start the server
mongoose
  .connect(process.env.MONGO_URI)
  .then(() => {
    console.log("MongoDB connected");
    app.listen(PORT, () => console.log(`Server running at http://localhost:${PORT}`));
  })
  .catch((err) => {
    console.error("MongoDB connection failed:", err.message);
    process.exit(1);
  });
