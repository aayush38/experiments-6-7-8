// Basic Node.js server using the built-in http module.
// Demonstrates: basic routing, serving HTML / JSON, and a callback (fs.readFile).
const http = require("http");
const fs = require("fs");
const path = require("path");

const PORT = 3000;
const booksFile = path.join(__dirname, "data", "books.json");

const server = http.createServer((req, res) => {
  console.log(`Request received: ${req.method} ${req.url}`);

  if (req.url === "/" || req.url === "/home") {
    // Route 1: simple HTML page
    res.writeHead(200, { "Content-Type": "text/html" });
    res.end(`
      <html>
        <head><title>Campus Library - Node HTTP Server</title></head>
        <body>
          <h1>Campus Library Catalogue</h1>
          <p>Basic Node.js http server is running.</p>
          <p>Try <a href="/api/books">/api/books</a> for the JSON catalogue.</p>
        </body>
      </html>`);
  } else if (req.url === "/api/books") {
    // Route 2: JSON data. fs.readFile takes a CALLBACK that runs when the file has been read.
    fs.readFile(booksFile, "utf-8", (err, data) => {
      if (err) {
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ error: "Could not read book data" }));
        return;
      }
      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(data);
    });
  } else {
    // Any other URL
    res.writeHead(404, { "Content-Type": "text/plain" });
    res.end("404 - Route not found");
  }
});

server.listen(PORT, () => {
  console.log(`Basic http server running at http://localhost:${PORT}/`);
});

2_file_sync.js

// Synchronous file operations: each call BLOCKS until it finishes.
const fs = require("fs");
const path = require("path");

const filePath = path.join(__dirname, "data", "issue_log_sync.json");
const record = { bookId: "LIB-101", issuedTo: "Student-55", time: new Date().toISOString() };

console.log("1. Writing file synchronously...");
fs.writeFileSync(filePath, JSON.stringify(record, null, 2));
console.log("2. Write finished.");

console.log("3. Reading file synchronously...");
const content = fs.readFileSync(filePath, "utf-8");
console.log("4. Read finished. Contents:");
console.log(content);

console.log("5. This line runs only after both operations complete (blocking).");

3_file_async.js

// Asynchronous file operations: calls return immediately, callbacks run later.
const fs = require("fs");
const path = require("path");

const filePath = path.join(__dirname, "data", "issue_log_async.json");
const record = { bookId: "LIB-103", issuedTo: "Student-56", time: new Date().toISOString() };

console.log("1. Starting asynchronous write...");
fs.writeFile(filePath, JSON.stringify(record, null, 2), (err) => {
  if (err) throw err;
  console.log("3. Write finished (callback fired).");

  console.log("4. Starting asynchronous read...");
  fs.readFile(filePath, "utf-8", (err, data) => {
    if (err) throw err;
    console.log("5. Read finished (callback fired). Contents:");
    console.log(data);
  });
});

console.log("2. This line runs immediately, before the write finishes.");

customModules/mathUtils.js

// Custom module containing reusable mathematical functions (CommonJS).
function add(a, b) { return a + b; }
function average(arr) { return arr.reduce((s, x) => s + x, 0) / arr.length; }
function factorial(n) { return n <= 1 ? 1 : n * factorial(n - 1); }
function isPrime(n) {
  if (n < 2) return false;
  for (let i = 2; i * i <= n; i++) if (n % i === 0) return false;
  return true;
}
function gcd(a, b) { return b === 0 ? a : gcd(b, a % b); }
function fibonacci(n) {
  const seq = [0, 1];
  for (let i = 2; i < n; i++) seq.push(seq[i - 1] + seq[i - 2]);
  return seq.slice(0, n);
}
// Library-specific use of the module: late fine = Rs. 2 per day after 7 free days
function lateFine(daysKept) { return daysKept > 7 ? (daysKept - 7) * 2 : 0; }

module.exports = { add, average, factorial, isPrime, gcd, fibonacci, lateFine };

expressApp/server.js

// Express app: routes + built-in static middleware + EJS template engine.
const express = require("express");
const path = require("path");
const fs = require("fs");
const mathUtils = require("../customModules/mathUtils");

const app = express();
const PORT = 3001;

// Built-in middleware: serve static files (CSS) from /public
app.use(express.static(path.join(__dirname, "public")));

// Template engine setup
app.set("view engine", "ejs");
app.set("views", path.join(__dirname, "views"));

const books = JSON.parse(
  fs.readFileSync(path.join(__dirname, "..", "data", "books.json"), "utf-8")
);
const findBook = (id) => books.find((b) => b.bookId === id);

// Dynamic page 1: catalogue
app.get("/", (req, res) => {
  res.render("index", { books });
});

// Dynamic page 2: details of one book (route parameter)
app.get("/books/:id", (req, res) => {
  const book = findBook(req.params.id);
  if (!book) return res.status(404).send("Book not found");
  const daysKept = Math.floor((Date.now() - new Date(book.issueDate)) / 86400000);
  res.render("book", { book, daysKept, fine: mathUtils.lateFine(daysKept) });
});

// JSON API
app.get("/api/books/:id", (req, res) => {
  const book = findBook(req.params.id);
  if (!book) return res.status(404).json({ error: "Book not found" });
  res.json(book);
});

app.listen(PORT, () => console.log(`Express app running at http://localhost:${PORT}/`));

// Basic routing using Express.js (GET / POST / all)
var express = require("express");
var app = express();

app.get("/hello", function (req, res) {
  res.send("Hello world");
});

app.post("/hello", function (req, res) {
  res.send("You just called the POST method at '/hello'!\n");
});

app.all("/test", function (req, res) {
  res.send("HTTP method doesn't have an effect on this route!");
});

app.listen(3000, () => console.log("Express basic routing on http://localhost:3000"));
